{{- /* Overrides @thulite/seo's article schema so it follows the build spec: headline is the search-bearing h1 rather than the conversational title, description is the summary, dateModified comes from lastReviewed rather than file mtime, and the author is a Person with sameAs to LinkedIn. The extra fields below are for generative engines, which lean on explicit section, keywords and length far more than classic search. */ -}} {{- $headline := .Params.h1 | default .Title -}} {{- $description := .Params.summary | default .Params.definition -}} {{- $modified := .Params.lastReviewed | default .Date -}} {{- $author := dict "@type" "Person" "name" site.Params.author "url" site.BaseURL -}} {{- with site.Params.linkedin }}{{ $author = merge $author (dict "sameAs" (slice .)) }}{{ end -}} {{- $publisher := dict "@type" "Organization" "name" site.Params.legalEntity "url" site.BaseURL -}} {{- with site.Params.linkedin }}{{ $publisher = merge $publisher (dict "sameAs" (slice .)) }}{{ end -}} {{- with resources.Get "favicon.png" -}} {{- $publisher = merge $publisher (dict "logo" (dict "@type" "ImageObject" "url" .Permalink)) -}} {{- end -}} {{- $article := dict "@type" "Article" "headline" $headline "description" $description "abstract" (.Params.definition | default $description) "isPartOf" (dict "@id" .Permalink) "mainEntityOfPage" (dict "@id" .Permalink) "url" .Permalink "inLanguage" (site.Language.Locale | default "en-GB") "datePublished" (.Date.Format "2006-01-02T15:04:05-07:00") "dateModified" (time.Format "2006-01-02T15:04:05-07:00" $modified) "wordCount" .WordCount "author" $author "publisher" $publisher "isAccessibleForFree" true -}} {{- with .Params.tags -}} {{- $article = merge $article (dict "keywords" (delimit . ", ")) -}} {{- end -}} {{- with .Params.categories -}} {{- $section := . -}} {{- with site.GetPage (printf "/categories/%s" $section) }}{{ $section = .Title }}{{ end -}} {{- $article = merge $article (dict "articleSection" $section) -}} {{- end -}} {{- with site.Params.images -}} {{- $article = merge $article (dict "image" (slice (index . 0 | absURL))) -}} {{- end -}}
